# The toolchain, single-sourced.
#
# build/docker/devcontainer/Dockerfile installs exactly this list, and every CI
# stage runs in that image, so a developer running `make docker-ci` and the gate
# running on a runner are using the same tools. Two lists would drift, and the
# drift would show up as a gate that passes locally and fails in CI for reasons
# nobody can reproduce.
#
# One package per line; # starts a comment.
#
# The Dockerfile splits it: a line whose comment begins `[toolchain]` is an
# analysis tool, installed in the `toolchain` stage only, so it never ships into
# `integration` (handbook devcontainer.md §1); every other line is the compiler
# and runtime, installed in `base`.

gcc-avr          # the AVR compiler
avr-libc         # its C library and device headers
binutils-avr     # assembler, linker, avr-size
avrdude          # flashing

simavr           # the emulator: models ADC, EEPROM, USART, SPI, TWI, timers,
libsimavr-dev    # GPIO with pin-change interrupts, and the watchdog
libelf-dev       # simavr reads the firmware .elf through libelf

build-essential  # the host compiler, for the unit tests and tfwsim
make

clang-format     # [toolchain] stands in for gofmt
clang-tidy       # [toolchain] stands in for golangci-lint
cppcheck         # [toolchain] stands in for go vet

gcovr            # [toolchain] coverage, with the 90% floor
lcov             # [toolchain]

python3

openssl          # tfwctl auth's external signer: parses/signs an operator
                 # key in its own process so the key never enters tfwctl's
